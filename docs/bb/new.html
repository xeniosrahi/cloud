Architecture Revision

Centralized Master Data + Centralized Ledgers

✅ Key Architectural Change

All Master Data Tables are CENTRALIZED (no origin_id)

  • • transporters - Shared across all factories
  • • parties - Shared across all factories
  • • petrol_pumps - Shared across all factories
  • • sessions - Shared fiscal years
  • • invoices - Centralized payment documents
  • • ledger_advances - ⚡ NOW CENTRALIZED (with optional metadata)
  • • ledger_commissions - ⚡ NOW CENTRALIZED (with optional metadata)

🔥 Origin Tracking at TRANSACTION Level

  • • sales table has origin_id + references to centralized masters
  • • invoice_line_items has origin_id per line

💡 Business Logic Example

Transporter: M.K. Motors (works for Wonder + Mangalam)

OLD DESIGN (per-factory ledgers):
├─ Wonder advances: ₹50,000 (origin_id=1)
├─ Mangalam advances: ₹30,000 (origin_id=2)

NEW DESIGN (centralized ledgers):
├─ Total advances: ₹80,000 (no origin_id)
├─ Total commissions: ₹5,000 (no origin_id)
└─ One invoice settles from unified account

Detailed Table Schemas

SQL CREATE statements for all core tables

1. ORIGINS (Tenant Master) 

CREATE TABLE origins (
    -- Primary Key
    origin_id INT AUTO_INCREMENT PRIMARY KEY,

    -- Identification
    origin_code VARCHAR(50) UNIQUE NOT NULL,
    origin_name VARCHAR(255) NOT NULL,
    legal_name VARCHAR(500),

    -- Tax & Legal
    gst_number VARCHAR(20),
    pan_number VARCHAR(20),

    -- Contact Information
    address TEXT,
    city VARCHAR(100),
    state VARCHAR(100),
    pincode VARCHAR(10),
    phone VARCHAR(15),
    email VARCHAR(255),

    -- Status & Timestamps
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    INDEX idx_origin_code (origin_code),
    INDEX idx_is_active (is_active)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

2. TRANSPORTERS NO origin_id

✨ Benefits: No duplicate transporter entries. Single source of truth. Can work across multiple factories.

CREATE TABLE transporters (
    -- Primary Key
    transporter_id INT AUTO_INCREMENT PRIMARY KEY,

    -- Basic Information
    name VARCHAR(255) UNIQUE NOT NULL,           -- ✨ UNIQUE across all factories
    gst VARCHAR(20),
    pan VARCHAR(20),
    address TEXT,
    phone VARCHAR(15),

    -- Banking Details
    bank_name VARCHAR(255),
    bank_account VARCHAR(50),
    ifsc_code VARCHAR(20),

    -- Status & Timestamps
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    -- Indexes
    INDEX idx_name (name),
    INDEX idx_is_active (is_active),
    INDEX idx_gst (gst)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ✨ Same transporter can work for multiple factories!
-- Relationship tracked in sales table via (origin_id + transporter_id)

3. LEDGER_ADVANCES CENTRALIZED

⚡ Key Changes: Removed origin_id. Added given_by_origin_id (optional metadata). Added settled_amount for partial settlements. One unified account per transporter.

CREATE TABLE ledger_advances (
    -- Primary Key
    advance_id INT AUTO_INCREMENT PRIMARY KEY,

    -- ⚡ NO origin_id - Centralized!

    -- Foreign Keys
    transporter_id INT NOT NULL,
    given_by_user_id INT,

    -- Advance Details
    advance_amount DECIMAL(10,2) NOT NULL,
    entry_date DATE NOT NULL,
    payment_mode ENUM('CASH', 'CHEQUE', 'RTGS', 'NEFT', 'UPI') DEFAULT 'CASH',
    payment_reference VARCHAR(100),

    -- Optional: Track which factory gave it (for reporting only)
    given_by_origin_id INT,

    -- Settlement Tracking
    is_settled BOOLEAN DEFAULT FALSE,
    settled_in_invoice_id INT,
    settled_date DATE,
    settled_amount DECIMAL(10,2),

    -- Description
    description TEXT,
    notes TEXT,

    -- Timestamps
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    -- Indexes
    INDEX idx_transporter (transporter_id),
    INDEX idx_is_settled (is_settled),
    INDEX idx_entry_date (entry_date),
    INDEX idx_given_by_origin (given_by_origin_id),

    FOREIGN KEY (transporter_id) REFERENCES transporters(transporter_id),
    FOREIGN KEY (given_by_user_id) REFERENCES users(user_id),
    FOREIGN KEY (given_by_origin_id) REFERENCES origins(origin_id),
    FOREIGN KEY (settled_in_invoice_id) REFERENCES invoices(invoice_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

4. SALES WITH origin_id

Key Points: origin_id identifies which factory. All master references (transporter, party, pump) are centralized. LR number unique per (origin + session).

CREATE TABLE sales (
    -- Primary Key
    sale_id INT AUTO_INCREMENT PRIMARY KEY,

    -- 🔥 Multi-Tenancy (origin tracking)
    origin_id INT NOT NULL,

    -- Foreign Keys (all point to CENTRALIZED masters)
    session_id INT NOT NULL,
    transporter_id INT NOT NULL,
    party_id INT NOT NULL,
    pump_id INT,

    -- Trip Identification
    lr_number INT NOT NULL,
    dispatch_date DATE NOT NULL,
    vehicle_number VARCHAR(50),
    factory_bill_no VARCHAR(100),
    destination VARCHAR(255),

    -- Cargo Details
    qty_mt DECIMAL(10,2) NOT NULL,
    freight_rate_per_mt DECIMAL(10,2) NOT NULL,

    -- Computed Freight
    total_freight DECIMAL(10,2) GENERATED ALWAYS AS 
        (qty_mt * freight_rate_per_mt) STORED,

    -- Trip Expenses
    advance_amount DECIMAL(10,2) DEFAULT 0,
    diesel_amount DECIMAL(10,2) DEFAULT 0,
    shortage_amount DECIMAL(10,2) DEFAULT 0,
    unloading_charges DECIMAL(10,2) DEFAULT 0,

    -- Computed Balance
    trip_balance DECIMAL(10,2) GENERATED ALWAYS AS (
        (qty_mt * freight_rate_per_mt + unloading_charges) - 
        (advance_amount + diesel_amount + shortage_amount)
    ) STORED,

    -- Payment Tracking
    payment_status ENUM('UNPAID', 'INVOICED', 'PAID') DEFAULT 'UNPAID',
    payment_date DATE,

    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    -- Indexes & Constraints
    INDEX idx_origin (origin_id),
    INDEX idx_session (session_id),
    INDEX idx_transporter (transporter_id),
    INDEX idx_party (party_id),
    INDEX idx_dispatch_date (dispatch_date),
    INDEX idx_payment_status (payment_status),

    -- Business Rule: LR number unique per origin+session
    UNIQUE KEY unique_lr_per_origin_session (origin_id, session_id, lr_number),

    FOREIGN KEY (origin_id) REFERENCES origins(origin_id),
    FOREIGN KEY (session_id) REFERENCES sessions(session_id),
    FOREIGN KEY (transporter_id) REFERENCES transporters(transporter_id),
    FOREIGN KEY (party_id) REFERENCES parties(party_id),
    FOREIGN KEY (pump_id) REFERENCES petrol_pumps(pump_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

💡 Note: Additional tables (parties, petrol_pumps, sessions, invoices, invoice_line_items, ledger_commissions, users, audit_logs) follow similar patterns. See complete documentation for all schemas.

Centralized Ledger Tables

Unified accounting - one account per transporter

⚡ Key Changes from Legacy

❌ Removed
  • • origin_id (no filtering by factory)
  • • Per-factory ledger accounts
  • • Complex settlement logic
✅ Added
  • • given_by_origin_id (optional metadata)
  • • settled_amount (partial settlements)
  • • One unified account per transporter

ledger_advances

Money given TO transporter

  • • No origin_id for filtering
  • • Optional: given_by_origin_id
  • • Tracks payment mode
  • • Settlement status
  • • Partial settlement support

ledger_commissions

Money received FROM transporter

  • • No origin_id for filtering
  • • Optional: received_by_origin_id
  • • Tracks payment mode
  • • Settlement status
  • • Partial settlement support

💡 Unified Account Calculation

-- Simple: One account per transporter
SELECT 
    t.name,

    -- Trip balances across ALL factories
    SUM(s.trip_balance) AS total_trips_outstanding,

    -- Unsettled advances (from ALL factories)
    (SELECT COALESCE(SUM(advance_amount), 0) 
     FROM ledger_advances 
     WHERE transporter_id = t.transporter_id 
       AND is_settled = FALSE) AS unsettled_advances,

    -- Unsettled commissions (from ALL factories)
    (SELECT COALESCE(SUM(commission_amount), 0) 
     FROM ledger_commissions 
     WHERE transporter_id = t.transporter_id 
       AND is_settled = FALSE) AS unsettled_commissions,

    -- Net payable
    SUM(s.trip_balance) - 
    (SELECT COALESCE(SUM(advance_amount), 0) ...) +
    (SELECT COALESCE(SUM(commission_amount), 0) ...) AS net_payable

FROM transporters t
LEFT JOIN sales s ON s.transporter_id = t.transporter_id 
    AND s.payment_status = 'UNPAID'
WHERE t.transporter_id = 1
GROUP BY t.transporter_id;

Common Query Patterns

Production-ready SQL queries for frequent operations

1. Transporter Working Across Multiple Factories

SELECT 
    t.name AS transporter,
    o.origin_name AS factory,
    COUNT(s.sale_id) AS total_trips,
    SUM(s.qty_mt) AS total_quantity_mt,
    SUM(s.total_freight) AS total_freight,
    SUM(s.trip_balance) AS outstanding_balance
FROM transporters t
CROSS JOIN origins o
LEFT JOIN sales s ON s.transporter_id = t.transporter_id 
    AND s.origin_id = o.origin_id
    AND s.payment_status != 'PAID'
WHERE t.transporter_id = 1
GROUP BY t.transporter_id, o.origin_id
ORDER BY o.origin_name;

2. Unified Account Statement

-- Complete account statement for a transporter
SELECT 
    'Trips Outstanding' AS category,
    o.origin_name AS origin,
    COUNT(s.sale_id) AS count,
    SUM(s.trip_balance) AS amount
FROM sales s
JOIN origins o ON o.origin_id = s.origin_id
WHERE s.transporter_id = 1 
  AND s.payment_status = 'UNPAID'
GROUP BY o.origin_id

UNION ALL

SELECT 
    'Unsettled Advances' AS category,
    o.origin_name AS origin,
    COUNT(la.advance_id) AS count,
    SUM(la.advance_amount) AS amount
FROM ledger_advances la
LEFT JOIN origins o ON o.origin_id = la.given_by_origin_id
WHERE la.transporter_id = 1 
  AND la.is_settled = FALSE
GROUP BY la.given_by_origin_id

UNION ALL

SELECT 
    'Unsettled Commissions' AS category,
    o.origin_name AS origin,
    COUNT(lc.commission_id) AS count,
    SUM(lc.commission_amount) AS amount
FROM ledger_commissions lc
LEFT JOIN origins o ON o.origin_id = lc.received_by_origin_id
WHERE lc.transporter_id = 1 
  AND lc.is_settled = FALSE
GROUP BY lc.received_by_origin_id

ORDER BY category, origin;

3. Net Payable Calculation

SELECT 
    t.name,

    -- Trip balances across ALL factories
    SUM(s.trip_balance) AS total_trips_outstanding,

    -- Unsettled advances
    (SELECT COALESCE(SUM(advance_amount), 0) 
     FROM ledger_advances 
     WHERE transporter_id = t.transporter_id 
       AND is_settled = FALSE) AS unsettled_advances,

    -- Unsettled commissions
    (SELECT COALESCE(SUM(commission_amount), 0) 
     FROM ledger_commissions 
     WHERE transporter_id = t.transporter_id 
       AND is_settled = FALSE) AS unsettled_commissions,

    -- Net payable
    SUM(s.trip_balance) - 
    (SELECT COALESCE(SUM(advance_amount), 0) 
     FROM ledger_advances 
     WHERE transporter_id = t.transporter_id 
       AND is_settled = FALSE) +
    (SELECT COALESCE(SUM(commission_amount), 0) 
     FROM ledger_commissions 
     WHERE transporter_id = t.transporter_id 
       AND is_settled = FALSE) AS net_payable

FROM transporters t
LEFT JOIN sales s ON s.transporter_id = t.transporter_id 
    AND s.payment_status = 'UNPAID'
WHERE t.transporter_id = 1
GROUP BY t.transporter_id;
